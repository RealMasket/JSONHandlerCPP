find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(AUTO_V2_SPEED_BUDGET 4194304 CACHE STRING "Speculative speed reserve bytes (geometric floor may override)")
set(AUTO_V2_MEMORY_BUDGET 524288 CACHE STRING "Speculative memory reserve bytes (geometric floor may override)")
foreach(value AUTO_V2_SPEED_BUDGET AUTO_V2_MEMORY_BUDGET)
    if(NOT "${${value}}" MATCHES "^[0-9]+$" OR ${value} GREATER 1073741824)
        message(FATAL_ERROR "${value} must be an integer in 0..1073741824")
    endif()
endforeach()
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generate_lexer.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_lexer.py"
    "${PROJECT_SOURCE_DIR}/parser/JsonLexer.cpp" "${generated}" COMMAND_ERROR_IS_FATAL ANY)
set(v2_programs)
foreach(goal speed memory)
    foreach(mode time trace profile)
        if(mode STREQUAL "profile" AND NOT TARGET ParserProfile)
            continue()
        endif()
        set(library AutoV2-${goal}-${mode})
        add_library(${library} STATIC "${generated}/${goal}.cpp" $<TARGET_OBJECTS:AutoParser>)
        target_link_libraries(${library} PUBLIC MasketJSON)
        target_include_directories(${library} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/benchmarks/auto" "${PROJECT_SOURCE_DIR}/benchmarks/reservation")
        target_compile_definitions(${library} PUBLIC JsonLexer=AutoLexer JsonParser=AutoParser
            AUTO_V2_SPEED_BYTES=${AUTO_V2_SPEED_BUDGET} AUTO_V2_MEMORY_BYTES=${AUTO_V2_MEMORY_BUDGET})
        if(mode STREQUAL "time")
            add_executable(AutoV2Time-${goal} ../reservation/Timing.cpp ../BenchmarkUtils.cpp)
            target_compile_definitions(AutoV2Time-${goal} PRIVATE MATRIX_CONFIGURATION="$<CONFIG>")
            target_link_libraries(AutoV2Time-${goal} PRIVATE ${library})
            add_executable(AutoV2Inspect-${goal} ../reservation/Inspect.cpp)
            target_link_libraries(AutoV2Inspect-${goal} PRIVATE ${library})
            add_executable(AutoV2Verify-${goal} Verify.cpp ../auto/Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
            target_include_directories(AutoV2Verify-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
            target_link_libraries(AutoV2Verify-${goal} PRIVATE ${library})
            add_executable(AutoV2Replay-${goal} "${PROJECT_SOURCE_DIR}/fuzz/FuzzReplay.cpp" ../auto/Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
            target_include_directories(AutoV2Replay-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
            target_link_libraries(AutoV2Replay-${goal} PRIVATE ${library})
            list(APPEND v2_programs AutoV2Time-${goal} AutoV2Inspect-${goal} AutoV2Verify-${goal} AutoV2Replay-${goal})
        elseif(mode STREQUAL "trace")
            target_compile_definitions(${library} PUBLIC MASKETJSON_AUTO_V2_TRACE)
            add_executable(AutoV2Trace-${goal} Trace.cpp)
            target_link_libraries(AutoV2Trace-${goal} PRIVATE ${library})
            list(APPEND v2_programs AutoV2Trace-${goal})
        else()
            target_compile_definitions(${library} PUBLIC MASKETJSON_RESERVATION_STATS)
            add_executable(AutoV2Profile-${goal} ../ParserProfile.cpp ../MemoryTracker.cpp)
            target_link_libraries(AutoV2Profile-${goal} PRIVATE ${library} Threads::Threads)
            list(APPEND v2_programs AutoV2Profile-${goal})
        endif()
    endforeach()
endforeach()
add_custom_target(AutoV2Tools DEPENDS ${v2_programs} AutoTime-balanced AutoTime-low-memory AutoInspect-balanced AutoInspect-low-memory)
if(TARGET ParserProfile)
    add_dependencies(AutoV2Tools AutoProfile-balanced AutoProfile-low-memory)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<IF:$<BOOL:${CMAKE_CONFIGURATION_TYPES}>,$<CONFIG>/,>build-info-$<CONFIG>.txt"
    CONTENT "compiler=${CMAKE_CXX_COMPILER_ID}\ncompiler_version=${CMAKE_CXX_COMPILER_VERSION}\nconfiguration=$<CONFIG>\nspeed_budget_bytes=${AUTO_V2_SPEED_BUDGET}\nmemory_budget_bytes=${AUTO_V2_MEMORY_BUDGET}\nconfig_flags=$<$<CONFIG:Release>:${CMAKE_CXX_FLAGS_RELEASE}>$<$<CONFIG:RelWithDebInfo>:${CMAKE_CXX_FLAGS_RELWITHDEBINFO}>\n")
if(MASKETJSON_BUILD_FUZZERS)
    foreach(goal speed memory)
        add_library(AutoV2-${goal}-fuzz STATIC "${generated}/${goal}.cpp" $<TARGET_OBJECTS:AutoParserFuzz>)
        target_link_libraries(AutoV2-${goal}-fuzz PUBLIC MasketJSON)
        target_include_directories(AutoV2-${goal}-fuzz PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/benchmarks/auto")
        target_compile_definitions(AutoV2-${goal}-fuzz PUBLIC JsonLexer=AutoLexer JsonParser=AutoParser
            AUTO_V2_SPEED_BYTES=${AUTO_V2_SPEED_BUDGET} AUTO_V2_MEMORY_BYTES=${AUTO_V2_MEMORY_BUDGET})
        target_compile_options(AutoV2-${goal}-fuzz PRIVATE -fsanitize=fuzzer-no-link)
        add_executable(AutoV2Fuzz-${goal} "${PROJECT_SOURCE_DIR}/fuzz/ParserFuzz.cpp" ../auto/Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
        target_include_directories(AutoV2Fuzz-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
        target_compile_options(AutoV2Fuzz-${goal} PRIVATE -fsanitize=fuzzer-no-link)
        target_link_options(AutoV2Fuzz-${goal} PRIVATE -fsanitize=fuzzer)
        target_link_libraries(AutoV2Fuzz-${goal} PRIVATE AutoV2-${goal}-fuzz)
    endforeach()
endif()
