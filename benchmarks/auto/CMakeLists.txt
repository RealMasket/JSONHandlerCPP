if(BUILD_SHARED_LIBS)
    message(FATAL_ERROR "Auto experiment requires a static build")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/parser/JsonLexer.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/generate_lexer.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_lexer.py"
    "${PROJECT_SOURCE_DIR}/parser/JsonLexer.cpp" "${generated}" COMMAND_ERROR_IS_FATAL ANY)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<IF:$<BOOL:${CMAKE_CONFIGURATION_TYPES}>,$<CONFIG>/,>build-info-$<CONFIG>.txt"
    CONTENT "compiler=${CMAKE_CXX_COMPILER_ID}\ncompiler_version=${CMAKE_CXX_COMPILER_VERSION}\nconfiguration=$<CONFIG>\ncxx_standard=${CMAKE_CXX_STANDARD}\nflags=${CMAKE_CXX_FLAGS}\nconfig_flags=$<$<CONFIG:Release>:${CMAKE_CXX_FLAGS_RELEASE}>$<$<CONFIG:Debug>:${CMAKE_CXX_FLAGS_DEBUG}>\nlto=${CMAKE_INTERPROCEDURAL_OPTIMIZATION}\n")

# Private names allow an independent production parser/lexer in the same process.
# Common JSON implementation objects are supplied by the normal static library.
add_library(AutoParser OBJECT "${PROJECT_SOURCE_DIR}/parser/JsonParser.cpp")
target_link_libraries(AutoParser PRIVATE MasketJSON)
target_compile_definitions(AutoParser PRIVATE JsonLexer=AutoLexer JsonParser=AutoParser)
add_library(AutoBaseline OBJECT Baseline.cpp)
target_link_libraries(AutoBaseline PRIVATE MasketJSON)

foreach(fixed balanced low-memory)
    add_executable(AutoTime-${fixed} ../reservation/Timing.cpp ../BenchmarkUtils.cpp)
    target_link_libraries(AutoTime-${fixed} PRIVATE MasketJSON)
    target_compile_definitions(AutoTime-${fixed} PRIVATE MATRIX_CONFIGURATION="$<CONFIG>")
    add_executable(AutoInspect-${fixed} ../reservation/Inspect.cpp)
    target_link_libraries(AutoInspect-${fixed} PRIVATE MasketJSON)
    if(fixed STREQUAL "low-memory")
        target_compile_definitions(AutoTime-${fixed} PRIVATE MATRIX_LOW_MEMORY)
        target_compile_definitions(AutoInspect-${fixed} PRIVATE MATRIX_LOW_MEMORY)
    endif()
endforeach()

foreach(goal speed memory)
    foreach(mode time trace profile)
        if(mode STREQUAL "profile" AND NOT TARGET ParserProfile)
            continue()
        endif()
        set(library Auto-${goal}-${mode})
        add_library(${library} STATIC "${generated}/${goal}.cpp" $<TARGET_OBJECTS:AutoParser>)
        target_link_libraries(${library} PUBLIC MasketJSON)
        target_include_directories(${library} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/benchmarks/reservation")
        target_compile_definitions(${library} PUBLIC JsonLexer=AutoLexer JsonParser=AutoParser)
        if(mode STREQUAL "time")
            add_executable(AutoTime-${goal} ../reservation/Timing.cpp ../BenchmarkUtils.cpp)
            target_compile_definitions(AutoTime-${goal} PRIVATE MATRIX_CONFIGURATION="$<CONFIG>")
            target_link_libraries(AutoTime-${goal} PRIVATE ${library})
            add_executable(AutoInspect-${goal} ../reservation/Inspect.cpp)
            target_link_libraries(AutoInspect-${goal} PRIVATE ${library})
            add_executable(AutoVerify-${goal} Verify.cpp Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
            target_include_directories(AutoVerify-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
            target_link_libraries(AutoVerify-${goal} PRIVATE ${library})
            add_executable(AutoReplay-${goal} "${PROJECT_SOURCE_DIR}/fuzz/FuzzReplay.cpp" Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
            target_include_directories(AutoReplay-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
            target_link_libraries(AutoReplay-${goal} PRIVATE ${library})
        elseif(mode STREQUAL "trace")
            target_compile_definitions(${library} PUBLIC MASKETJSON_AUTO_TRACE)
            add_executable(AutoTrace-${goal} Trace.cpp)
            target_link_libraries(AutoTrace-${goal} PRIVATE ${library})
        else()
            target_compile_definitions(${library} PUBLIC MASKETJSON_RESERVATION_STATS)
            add_executable(AutoProfile-${goal} ../ParserProfile.cpp ../MemoryTracker.cpp)
            target_link_libraries(AutoProfile-${goal} PRIVATE ${library} Threads::Threads)
        endif()
    endforeach()
endforeach()

if(TARGET ParserProfile)
    add_library(Auto-fixed-profile STATIC "${generated}/fixed.cpp" $<TARGET_OBJECTS:AutoParser>)
    target_link_libraries(Auto-fixed-profile PUBLIC MasketJSON)
    target_include_directories(Auto-fixed-profile PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/benchmarks/reservation")
    target_compile_definitions(Auto-fixed-profile PUBLIC JsonLexer=AutoLexer JsonParser=AutoParser MASKETJSON_RESERVATION_STATS)
    foreach(fixed balanced low-memory)
        add_executable(AutoProfile-${fixed} ../ParserProfile.cpp ../MemoryTracker.cpp)
        target_link_libraries(AutoProfile-${fixed} PRIVATE Auto-fixed-profile Threads::Threads)
        if(fixed STREQUAL "low-memory")
            target_compile_definitions(AutoProfile-${fixed} PRIVATE MATRIX_LOW_MEMORY)
        endif()
    endforeach()
endif()

if(MASKETJSON_BUILD_FUZZERS)
    add_library(AutoParserFuzz OBJECT "${PROJECT_SOURCE_DIR}/parser/JsonParser.cpp")
    target_link_libraries(AutoParserFuzz PRIVATE MasketJSON)
    target_compile_definitions(AutoParserFuzz PRIVATE JsonLexer=AutoLexer JsonParser=AutoParser)
    target_compile_options(AutoParserFuzz PRIVATE -fsanitize=fuzzer-no-link)
    foreach(goal speed memory)
        add_library(Auto-${goal}-fuzz STATIC "${generated}/${goal}.cpp" $<TARGET_OBJECTS:AutoParserFuzz>)
        target_link_libraries(Auto-${goal}-fuzz PUBLIC MasketJSON)
        target_include_directories(Auto-${goal}-fuzz PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_definitions(Auto-${goal}-fuzz PUBLIC JsonLexer=AutoLexer JsonParser=AutoParser)
        target_compile_options(Auto-${goal}-fuzz PRIVATE -fsanitize=fuzzer-no-link)
        add_executable(AutoFuzz-${goal} "${PROJECT_SOURCE_DIR}/fuzz/ParserFuzz.cpp" Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
        target_include_directories(AutoFuzz-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
        target_compile_options(AutoFuzz-${goal} PRIVATE -fsanitize=fuzzer-no-link)
        target_link_options(AutoFuzz-${goal} PRIVATE -fsanitize=fuzzer)
        target_link_libraries(AutoFuzz-${goal} PRIVATE Auto-${goal}-fuzz)
    endforeach()
endif()
