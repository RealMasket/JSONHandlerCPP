std::string Json::serializeObject(const JsonObject& obj) const {
    std::string result = "{";
    bool first = true;

    for (const auto& [key, value] : obj) {
        if (!first) result += ",";
        result += "\"" + escapeString(key) + "\": ";
        
        result += value.serialize(false, 1);
        first = false;
    }

    result += "}";
    return result;
}
